> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A single Go binary that takes a pasted curl command, runs a staged load ramp (100 → 1M requests, rising concurrency) against the target, stores every response time in SQLite, and serves a live local dashboard with exact percentiles, saturation detection, kill switch, and run history.

**Architecture:** Go binary embedding a vanilla-JS UI (`go:embed` + vendored uPlot). Vegeta library is the request engine behind a thin wrapper; an orchestrator runs stages sequentially with abort/kill logic; results stream to SQLite (batched) and to the browser over SSE.

**Tech Stack:** Go 1.22+, `github.com/tsenart/vegeta/v12` (engine), `modernc.org/sqlite` (pure-Go SQLite), `github.com/google/shlex` (curl lexing), uPlot (vendored JS charts). No frontend build step.

```go
package curlparse

import (
	"encoding/base64"
	"fmt"
	"net/http"
	"strings"

	"github.com/google/shlex"
)

func Parse(cmd string) (Target, error) {
	toks, err := shlex.Split(strings.TrimSpace(cmd))
	if err != nil {
		return Target{}, fmt.Errorf("cannot parse command: %w", err)
	}
	if len(toks) > 0 && toks[0] == "curl" {
		toks = toks[1:]
	}
	t := Target{Header: http.Header{}}
	dataSeen := false
	next := func(i int, flag string) (string, error) {
		if i+1 >= len(toks) {
			return "", fmt.Errorf("flag %s expects a value", flag)
		}
		return toks[i+1], nil
	}
	for i := 0; i < len(toks); i++ {
		tok := toks[i]
		switch {
		case tok == "-X" || tok == "--request":
			v, err := next(i, tok)
			if err != nil {
				return Target{}, err
			}
			t.Method = strings.ToUpper(v)
			i++
		case tok == "-H" || tok == "--header":
			v, err := next(i, tok)
			if err != nil {
				return Target{}, err
			}
			k, val, ok := strings.Cut(v, ":")
			if !ok {
				return Target{}, fmt.Errorf("malformed header %q", v)
			}
			t.Header.Add(strings.TrimSpace(k), strings.TrimSpace(val))
			i++
		case tok == "-d" || tok == "--data" || tok == "--data-raw" || tok == "--data-binary":
			v, err := next(i, tok)
			if err != nil {
				return Target{}, err
			}
			t.Body = []byte(v)
			dataSeen = true
			i++
		case tok == "-u" || tok == "--user":
			v, err := next(i, tok)
			if err != nil {
				return Target{}, err
			}
			t.Header.Set("Authorization", "Basic "+base64.StdEncoding.EncodeToString([]byte(v)))
			i++
		case tok == "-k" || tok == "--insecure":
			t.Insecure = true
		case tok == "--url":
			v, err := next(i, tok)
			if err != nil {
				return Target{}, err
			}
			t.URL = v
			i++
		case strings.HasPrefix(tok, "-"):
			return Target{}, fmt.Errorf("flag %s not supported yet", tok)
		default:
			if t.URL != "" {
				return Target{}, fmt.Errorf("unexpected argument %q", tok)
			}
			t.URL = tok
		}
	}
	if t.URL == "" {
		return Target{}, fmt.Errorf("no URL found in command")
	}
	if !strings.HasPrefix(t.URL, "http://") && !strings.HasPrefix(t.URL, "https://") {
		t.URL = "http://" + t.URL
	}
	if t.Method == "" {
		if dataSeen {
			t.Method = "POST"
		} else {
			t.Method = "GET"
		}
	}
	return t, nil
}
```

```go
package metrics

import (
	"fmt"
	"math"
	"slices"
	"time"
)

// (types from Interfaces block above)

func DefaultPlan() []Stage {
	return []Stage{{100, 10}, {1000, 25}, {10000, 50}, {100000, 100}, {1000000, 250}}
}

func DefaultAbortRules() AbortRules { return AbortRules{MaxErrorRate: 0.5, MaxP99Ms: 5000} }

func Percentile(sorted []int64, q float64) int64 {
	if len(sorted) == 0 {
		return 0
	}
	idx := int(math.Ceil(q*float64(len(sorted)))) - 1
	if idx < 0 {
		idx = 0
	}
	if idx >= len(sorted) {
		idx = len(sorted) - 1
	}
	return sorted[idx]
}

type StageAcc struct {
	latencies    []int64
	codes        map[int]int
	errs         int
	timeouts     int
	sampleErrors []string
	seenErrors   map[string]bool
}

func NewStageAcc() *StageAcc {
	return &StageAcc{codes: map[int]int{}, seenErrors: map[string]bool{}}
}

func (a *StageAcc) Add(s Sample) {
	a.latencies = append(a.latencies, s.LatencyUs)
	if s.Error != "" {
		a.errs++
		if s.Timeout {
			a.timeouts++
		}
		if len(a.sampleErrors) < 5 && !a.seenErrors[s.Error] {
			a.seenErrors[s.Error] = true
			a.sampleErrors = append(a.sampleErrors, s.Error)
		}
		return
	}
	a.codes[s.Code]++
}

func (a *StageAcc) Count() int { return len(a.latencies) }

func (a *StageAcc) RollingP99Us() int64 {
	n := len(a.latencies)
	win := 5000
	if n < win {
		win = n
	}
	recent := slices.Clone(a.latencies[n-win:])
	slices.Sort(recent)
	return Percentile(recent, 0.99)
}

func (a *StageAcc) Summarize(stageIdx int, st Stage, wall time.Duration) StageSummary {
	sorted := slices.Clone(a.latencies)
	slices.Sort(sorted)
	total := len(sorted)
	var sum int64
	for _, v := range sorted {
		sum += v
	}
	s := StageSummary{
		Stage: stageIdx, Requests: st.Requests, Concurrency: st.Concurrency,
		DurationMs: wall.Milliseconds(),
		Codes:      map[string]int{}, ErrorCount: a.errs, Timeouts: a.timeouts,
		SampleErrors: a.sampleErrors,
	}
	if total > 0 {
		s.MinUs, s.MaxUs = sorted[0], sorted[total-1]
		s.MeanUs = sum / int64(total)
		s.P50Us = Percentile(sorted, 0.50)
		s.P90Us = Percentile(sorted, 0.90)
		s.P95Us = Percentile(sorted, 0.95)
		s.P99Us = Percentile(sorted, 0.99)
		s.ErrorRate = float64(a.errs) / float64(total)
	}
	if wall > 0 {
		s.RPS = float64(total) / wall.Seconds()
	}
	for code, n := range a.codes {
		s.Codes[fmt.Sprintf("%d", code)] = n
	}
	return s
}

func ShouldAbort(s StageSummary, r AbortRules) (bool, string) {
	if s.ErrorRate > r.MaxErrorRate {
		return true, fmt.Sprintf("error rate %.0f%% exceeds %.0f%%", s.ErrorRate*100, r.MaxErrorRate*100)
	}
	if s.P99Us > r.MaxP99Ms*1000 {
		return true, fmt.Sprintf("p99 %.1fs exceeds %.1fs", float64(s.P99Us)/1e6, float64(r.MaxP99Ms)/1e3)
	}
	return false, ""
}

func ComputeVerdict(sums []StageSummary, abortReason string) Verdict {
	v := Verdict{AbortReason: abortReason}
	if len(sums) == 0 {
		return v
	}
	baseP99 := sums[0].P99Us
	for i, s := range sums {
		if s.ErrorRate > 0.01 || (i > 0 && baseP99 > 0 && s.P99Us >= 4*baseP99) {
			idx := i
			v.SaturationStage = &idx
			v.Notes = append(v.Notes, fmt.Sprintf(
				"degradation first observed at stage %d (%d requests, concurrency %d)",
				i, s.Requests, s.Concurrency))
			break
		}
	}
	healthy := sums
	if v.SaturationStage != nil {
		healthy = sums[:*v.SaturationStage]
	}
	for _, s := range healthy {
		if s.RPS > v.MaxHealthyRPS {
			v.MaxHealthyRPS = s.RPS
		}
	}
	for i := 1; i < len(healthy); i++ {
		prev, cur := healthy[i-1], healthy[i]
		if cur.RPS < prev.RPS*1.10 && cur.ErrorRate <= 0.01 &&
			(prev.P99Us == 0 || cur.P99Us < prev.P99Us*3/2) {
			v.GeneratorLimited = true
			v.Notes = append(v.Notes,
				"throughput plateaued while the API stayed healthy — the load generator (this machine) may be the bottleneck; results above this RPS reflect client limits, not the API")
			break
		}
	}
	if v.SaturationStage == nil && !v.GeneratorLimited {
		v.Notes = append(v.Notes, "no degradation detected across all completed stages")
	}
	return v
}
```

```go
// InsertResults: one transaction + prepared statement.
func (s *Store) InsertResults(runID int64, stage int, batch []metrics.Sample) error {
	tx, err := s.db.Begin()
	if err != nil {
		return err
	}
	stmt, err := tx.Prepare(`INSERT INTO results(run_id,stage,offset_ms,latency_us,code,error) VALUES(?,?,?,?,?,?)`)
	if err != nil {
		tx.Rollback()
		return err
	}
	for _, r := range batch {
		if _, err := stmt.Exec(runID, stage, r.OffsetMs, r.LatencyUs, r.Code, r.Error); err != nil {
			tx.Rollback()
			return err
		}
	}
	return tx.Commit()
}

const seriesQ = `

SELECT MIN(offset_ms), CAST(AVG(latency_us) AS INTEGER), MAX(latency_us), COUNT(*)
FROM results, span WHERE run_id=? AND stage=?
GROUP BY offset_ms * ? / span.s ORDER BY 1`

           FROM results WHERE run_id=? AND stage=?)
SELECT b.lo + ((latency_us - b.lo)/b.w)*b.w AS from_us, COUNT(*)
FROM results, b WHERE run_id=? AND stage=?
```

```go
import (
	"context"
	"crypto/tls"
	"io"
	"net"
	"net/http"
	"net/url"
	"strings"
	"sync/atomic"
	"time"

	vegeta "github.com/tsenart/vegeta/v12/lib"

	"apiperf/curlparse"
	"apiperf/metrics"
)

func Run(ctx context.Context, t curlparse.Target, n, c int, timeout time.Duration) <-chan metrics.Sample {
	out := make(chan metrics.Sample, 1024)

	vt := vegeta.Target{Method: t.Method, URL: t.URL, Body: t.Body, Header: t.Header}
	var issued int64
	targeter := func(tgt *vegeta.Target) error {
		if atomic.AddInt64(&issued, 1) > int64(n) {
			return vegeta.ErrNoTargets
		}
		*tgt = vt
		return nil
	}

	opts := []func(*vegeta.Attacker){
		vegeta.Timeout(timeout),
		vegeta.Workers(uint64(c)),
		vegeta.MaxWorkers(uint64(c)),
		vegeta.KeepAlive(true),
		vegeta.Connections(c),
	}
	if t.Insecure {
		opts = append(opts, vegeta.TLSConfig(&tls.Config{InsecureSkipVerify: true}))
	}
	attacker := vegeta.NewAttacker(opts...)

	go func() {
		defer close(out)
		start := time.Now()
		results := attacker.Attack(targeter, vegeta.Rate{Freq: 0, Per: 0}, 0, "apiperf")
		done := make(chan struct{})
		go func() {
			select {
			case <-ctx.Done():
				attacker.Stop()
			case <-done:
			}
		}()
		defer close(done)

		sent := 0
		for res := range results {
			if res.Error == vegeta.ErrNoTargets.Error() {
				continue
			}
			out <- metrics.Sample{
				OffsetMs:  res.Timestamp.Sub(start).Milliseconds(),
				LatencyUs: res.Latency.Microseconds(),
				Code:      int(res.Code),
				Error:     res.Error,
				Timeout:   res.Error != "" && (strings.Contains(res.Error, "timeout") || strings.Contains(res.Error, "deadline")),
			}
			sent++
			if sent >= n {
				attacker.Stop()
			}
		}
	}()
	return out
}

func Probe(t curlparse.Target, timeout time.Duration) ProbeResult {
	client := &http.Client{Timeout: timeout}
	if t.Insecure {
		client.Transport = &http.Transport{TLSClientConfig: &tls.Config{InsecureSkipVerify: true}}
	}
	req, err := http.NewRequest(t.Method, t.URL, strings.NewReader(string(t.Body)))
	if err != nil {
		return ProbeResult{Error: err.Error()}
	}
	req.Header = t.Header.Clone()
	start := time.Now()
	res, err := client.Do(req)
	if err != nil {
		return ProbeResult{Error: err.Error(), LatencyMs: time.Since(start).Milliseconds()}
	}
	defer res.Body.Close()
	body, _ := io.ReadAll(io.LimitReader(res.Body, 64*1024))
	snip := body
	if len(snip) > 512 {
		snip = snip[:512]
	}
	return ProbeResult{
		Status: res.StatusCode, LatencyMs: time.Since(start).Milliseconds(),
		Bytes: len(body), Snippet: string(snip),
	}
}

func IsPrivateHost(rawURL string) bool {
	u, err := url.Parse(rawURL)
	if err != nil {
		return false
	}
	host := u.Hostname()
	if ip := net.ParseIP(host); ip != nil {
		return ip.IsLoopback() || ip.IsPrivate() || ip.IsLinkLocalUnicast()
	}
	ips, err := net.LookupIP(host)
	if err != nil || len(ips) == 0 {
		return false
	}
	for _, ip := range ips {
		if !(ip.IsLoopback() || ip.IsPrivate() || ip.IsLinkLocalUnicast()) {
			return false
		}
	}
	return true
}
```

```go
func (o *Orchestrator) Start(runID int64, tgt curlparse.Target, plan []metrics.Stage, rules metrics.AbortRules) error {
	if !o.running.CompareAndSwap(false, true) {
		return ErrBusy
	}
	ctx, cancel := context.WithCancel(context.Background())
	o.cancel = cancel
	o.activeID.Store(runID)
	go o.run(ctx, runID, tgt, plan, rules)
	return nil
}

func (o *Orchestrator) run(ctx context.Context, runID int64, tgt curlparse.Target, plan []metrics.Stage, rules metrics.AbortRules) {
	defer o.running.Store(false)
	o.st.SetStatus(runID, "running")
	status, abortReason := "completed", ""
	var sums []metrics.StageSummary

stages:
	for i, st := range plan {
		o.emit(Event{"stage_start", map[string]any{"stage": i, "requests": st.Requests, "concurrency": st.Concurrency}})
		acc := metrics.NewStageAcc()
		stageStart := time.Now()
		lastEmit := stageStart
		var buf []metrics.Sample

		for s := range engine.Run(ctx, tgt, st.Requests, st.Concurrency, o.timeout) {
			acc.Add(s)
			buf = append(buf, s)
			if len(buf) >= 1000 {
				o.st.InsertResults(runID, i, buf)
				buf = buf[:0]
			}
			if time.Since(lastEmit) >= 500*time.Millisecond {
				lastEmit = time.Now()
				o.emit(Event{"progress", map[string]any{
					"stage": i, "done": acc.Count(), "total": st.Requests,
					"rps":   float64(acc.Count()) / time.Since(stageStart).Seconds(),
					"p99Ms": float64(acc.RollingP99Us()) / 1000,
				}})
			}
		}
		if len(buf) > 0 {
			o.st.InsertResults(runID, i, buf)
		}
		sum := acc.Summarize(i, st, time.Since(stageStart))
		o.st.SaveStageSummary(runID, sum)
		o.emit(Event{"stage_done", sum})
		sums = append(sums, sum)

		if ctx.Err() != nil {
			status = "killed"
			break stages
		}
		if abort, reason := metrics.ShouldAbort(sum, rules); abort {
			status, abortReason = "aborted", reason
			break stages
		}
		if i < len(plan)-1 {
			select {
			case <-ctx.Done():
				status = "killed"
				break stages
			case <-time.After(o.cooldown):
			}
		}
	}
	v := metrics.ComputeVerdict(sums, abortReason)
	o.st.FinishRun(runID, status, v)
	o.emit(Event{"run_done", map[string]any{"status": status, "verdict": v}})
}
```

```go
import (

// Full loop: create → start → poll to completion → verify exact counts,
// summaries, verdict, series — through the real HTTP surface.
func TestFullRamp(t *testing.T) {
	var slow atomic.Bool
	var hits atomic.Int64
	target := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		hits.Add(1)
		if slow.Load() {
			time.Sleep(80 * time.Millisecond) // degrade in later stage
		}
		w.WriteHeader(200)
	}))
	defer target.Close()

	st, err := store.Open(filepath.Join(t.TempDir(), "e2e.db"))
	if err != nil {
		t.Fatal(err)
	}
	defer st.Close()
	hub := server.NewHub()
	o := orch.New(st, hub.Emit)
	o.SetCooldown(10 * time.Millisecond)
	app := httptest.NewServer(server.New(st, o, hub))
	defer app.Close()

	payload := fmt.Sprintf(`{"curl":"curl %s","stages":[{"requests":100,"concurrency":2},{"requests":500,"concurrency":5}]}`, target.URL)
	res, _ := http.Post(app.URL+"/api/runs", "application/json", strings.NewReader(payload))
	var created struct {
		RunID int64 `json:"runId"`
	}
	json.NewDecoder(res.Body).Decode(&created)

	res, _ = http.Post(fmt.Sprintf("%s/api/runs/%d/start", app.URL, created.RunID), "application/json", strings.NewReader(`{}`))
	if res.StatusCode != 200 {
		b, _ := io.ReadAll(res.Body)
		t.Fatalf("start: %d %s", res.StatusCode, b)
	}

	var detail store.RunDetail
	deadline := time.Now().Add(30 * time.Second)
	for {
		detail, err = st.GetRun(created.RunID)
		if err == nil && detail.Status == "completed" {
			break
		}
		if time.Now().After(deadline) {
			t.Fatalf("status = %s, never completed", detail.Status)
		}
		time.Sleep(50 * time.Millisecond)
	}

	if hits.Load() != 600 {
		t.Errorf("target hits = %d, want 600", hits.Load())
	}
	if len(detail.Summaries) != 2 {
		t.Fatalf("summaries = %d", len(detail.Summaries))
	}
	s0, s1 := detail.Summaries[0], detail.Summaries[1]
	if s0.Codes["200"] != 100 || s1.Codes["200"] != 500 {
		t.Errorf("codes: %+v / %+v", s0.Codes, s1.Codes)
	}
	if s0.P50Us <= 0 || s0.P99Us < s0.P50Us {
		t.Errorf("percentile sanity: %+v", s0)
	}
	if detail.Verdict == nil {
		t.Fatal("verdict missing")
	}

	var pts []store.SeriesPoint
	res, _ = http.Get(fmt.Sprintf("%s/api/runs/%d/series?stage=1", app.URL, created.RunID))
	json.NewDecoder(res.Body).Decode(&pts)
	var total int64
	for _, p := range pts {
		total += p.Count
	}
	if total != 500 {
		t.Errorf("stage 1 series total = %d, want 500", total)
	}
}
```

# apiperf

Paste a curl command, get the full performance story of your API: a staged
load ramp (100 → 1M requests, rising concurrency), exact p50/p90/p95/p99 per
stage, error breakdown, throughput, saturation-point detection, live
dashboard, kill switch, and run history — from one Go binary.

## Run

    go build -o apiperf . && ./apiperf
    # open http://localhost:7071

Paste your curl (supported flags: -X, -H, -d/--data*, -u, -k), hit Verify,
tune the stage plan if you like, Start. Results stream live; the final
report includes charts and a verdict. Every response time is stored in
`apiperf.db` (SQLite); the History tab reopens any past run.

## Notes

- Auto-abort: a run stops early when a stage exceeds the error-rate or p99
  thresholds (editable in the UI).
- Only load-test APIs you own or have permission to test. Non-private hosts
  require an explicit confirmation.
- At high concurrency the generator machine can be the bottleneck; the
  verdict flags this when throughput plateaus while the API stays healthy.
  Raise `ulimit -n` if you see file-descriptor errors (the binary attempts
  to raise its own soft limit at startup).
- A full 1M-request ramp stores ~1.1M rows (~40 MB) per run.
